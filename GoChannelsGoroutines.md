```go
wg := sync.WaitGroup{}
for i := range 5 {
	wg.Add(1)
	go func() {
		defer wg.Done()
		fmt.Println(i)
	}()
	wg.Wait()
}
```

```go
worker := func() chan int {
	ch := make(chan int)
	go func() {
		time.Sleep(3 * time.Second)
		ch <- 1
	}()
	return ch
}
t := time.Now()
_, _ = <-worker(), <-worker()
fmt.Println(int(time.Since(t).Seconds()))
```
